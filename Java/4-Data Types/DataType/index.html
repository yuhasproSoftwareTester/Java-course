    Data Types (Complete Guide)

    Java Data Types - 2 Categories

    1. Primitive (8 types - Built-in)

    Type Size Range Default Example
    byte 1 byte -128 to 127 0 byte b = 100;
    short 2 bytes -32,768 to 32,767 0 short s = 1000;
    int 4 bytes -2^31 to 2^31-1 0 int age = 25;
    long 8 bytes -2^63 to 2^63-1 0L long population = 100L;
    float 4 bytes 6-7 decimal 0.0f float pi = 3.14f;
    double 8 bytes 15 decimal 0.0d double salary = 50000.50;
    char 2 bytes Unicode '\u0000' char grade = 'A';
    boolean 1 bit true/false false boolean isValid = true;

    2. Non-Primitive (Reference)

    • String
    • Array
    • Classes
    • Objects

    Complete Examples - All 8 Primitives

    public class DataTypesDemo {
    public static void main(String[] args) {
        // All 8 primitive types
        byte b = 100;
        short s = 1000;
        int i = 25000;
        long l = 100000L;
        float f = 3.14f;
        double d = 3.14159;
        char c = 'A';
        boolean bool = true;

        // Print all
        System.out.println("byte: " + b);
        System.out.println("short: " + s);
        System.out.println("int: " + i);
        System.out.println("long: " + l);
        System.out.println("float: " + f);
        System.out.println("double: " + d);
        System.out.println("char: " + c);
        System.out.println("boolean: " + bool);
    }
}

    Memory & Range Comparison

    public class MemoryDemo {
    public static void main(String[] args) {
        // Same value, different types
        byte small = 100;           // 1 byte
        short medium = 100;         // 2 bytes  
        int normal = 100;           // 4 bytes (most common)
        long big = 100L;            // 8 bytes

        System.out.println("All store 100 but different memory:");
        System.out.println("byte: " + small);
        System.out.println("int: " + normal);
        System.out.println("long: " + big);
    }
}

    Practice 1: Student Record

    public class Practice1 {
    public static void main(String[] args) {
        String name = "Alice";
        int age = 20;
        char grade = 'A';
        double gpa = 8.75;
        boolean passed = true;

        System.out.println("Student: " + name);
        System.out.println("Age: " + age);

}

    Practice 2: Bank Account

    public class Practice2 {
    public static void main(String[] args) {
        String account = "ACC123";
        long balance = 50000L;
        double interest = 6.5;
        boolean active = true;
        char type = 'S';  // S=Savings, C=Current

        System.out.printf("Account: %s\n", account);
        System.out.printf("Balance: Rs. %d\n", balance);
        System.out.printf("Interest: %.1f%%\n", interest);
        System.out.printf("Type: %c\n", type);
    }
}

    Practice 3: Temperature Converter

    public class Practice3 {
    public static void main(String[] args) {

}

    ⚠️ Common Errors

    // ❌ Too big for type
byte b = 200;  // Error! max=127

// ❌ Missing L for long
long l = 10000000000;  // Error!

// ❌ Missing f for float
float f = 3.14;  // Error!

// ✅ Correct
byte b = 100;
long l = 10000000000L;
float f = 3.14f;

    Data Type Rules

    int = Most common (4 bytes)
long = Big numbers (add L)
float = Decimal (add f)
String = Text (capital S)
boolean = true/false only
char = Single quote 'A'
